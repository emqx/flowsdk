9pub(super) fn require(valid: bool, message: &str) -> Result<(), ParseError> {
10    if valid {
11        Ok(())
12    } else {
13        Err(ParseError::ParseError(message.to_owned()))
14    }
15}

17pub(super) fn string(value: &str) -> Result<(), ParseError> {
18    if value.len() > u16::MAX as usize {
19        return Err(ParseError::StringTooLong);
20    }
21    super::validate_mqtt_utf8_string(value)
22}

24pub(super) fn topic(value: &str) -> Result<(), ParseError> {
25    string(value)?;
26    require(!value.is_empty(), "Topic name cannot be empty")?;
27    require(
28        !value.contains(['+', '#']),
29        "Topic name cannot contain wildcards",
30    )
31}
32
33pub(super) fn packet_id(value: u16) -> Result<(), ParseError> {
34    require(value != 0, "Packet identifier cannot be zero")
35}
36
37pub(super) fn publish(qos: u8, dup: bool, id: Option<u16>) -> Result<(), ParseError> {
38    require(qos <= 2, "PUBLISH QoS cannot exceed 2")?;
39    require(qos != 0 || !dup, "PUBLISH DUP must be zero for QoS 0")?;
40    if qos > 0 {
41        packet_id(id.ok_or_else(|| ParseError::ParseError("Missing packet identifier".into()))?)?;
42    }
43    Ok(())
44}
45
46pub(super) fn connect_flags(flags: u8) -> Result<(), ParseError> {
47    require(flags & 1 == 0, "CONNECT reserved flag must be zero")?;
48    require((flags >> 3) & 3 != 3, "Will QoS cannot exceed 2")?;
49    require(
50        flags & 4 != 0 || flags & 0x38 == 0,
51        "Will QoS and RETAIN require the Will flag",
52    )
53}
54
55pub(super) fn fixed_header(byte: u8) -> Result<(), ParseError> {
56    let kind = byte >> 4;
57    let flags = byte & 0x0f;
58    let valid = match kind {
59        3 => {
60            let qos = (flags >> 1) & 3;
61            qos <= 2 && (qos != 0 || flags & 8 == 0)
62        }
63        6 | 8 | 10 => flags == 2,
64        1..=15 => flags == 0,
65        _ => false,
66    };
67    require(valid, "Invalid MQTT fixed header flags")
68}

83pub(super) fn property_value(property: &Property) -> Result<(), ParseError> {
84    use Property::*;
85    match property {
86        PayloadFormatIndicator(v)
87        | RequestProblemInformation(v)
88        | RequestResponseInformation(v)
89        | MaximumQoS(v)
90        | RetainAvailable(v)
91        | WildcardSubscriptionAvailable(v)
92        | SubscriptionIdentifierAvailable(v)
93        | SharedSubscriptionAvailable(v) => require(*v <= 1, "Property value must be 0 or 1"),
94        ReceiveMaximum(v) | TopicAlias(v) => require(*v != 0, "Property value cannot be zero"),
95        MaximumPacketSize(v) => require(*v != 0, "Maximum Packet Size cannot be zero"),
96        SubscriptionIdentifier(v) => require(
97            (1..=268_435_455).contains(v),
98            "Invalid Subscription Identifier",
99        ),
100        ResponseTopic(v) => topic(v),
101        AssignedClientIdentifier(v) => {
102            string(v)?;
103            require(!v.is_empty(), "Assigned Client Identifier cannot be empty")
104        }
105        ContentType(v)
106        | AuthenticationMethod(v)
107        | ResponseInformation(v)
108        | ServerReference(v)
109        | ReasonString(v) => string(v),
110        UserProperty(k, v) => {
111            string(k)?;
112            string(v)
113        }
114        CorrelationData(v) | AuthenticationData(v) => {
115            if v.len() > u16::MAX as usize {
116                Err(ParseError::StringTooLong)
117            } else {
118                Ok(())
119            }
120        }
121        _ => Ok(()),
122    }
123}

125pub(super) fn properties(
126    properties: &[Property],
127    context: PropertyContext,
128) -> Result<(), ParseError> {
129    use Property::*;
130    use PropertyContext as C;
131    // A bit per property identifier avoids allocations and quadratic duplicate checks.
132    let mut seen = 0u64;
133    for property in properties {
134        let (id, allowed) = match property {
135            PayloadFormatIndicator(_) => (0x01, matches!(context, C::Publish | C::Will)),
136            MessageExpiryInterval(_) => (0x02, matches!(context, C::Publish | C::Will)),
137            ContentType(_) => (0x03, matches!(context, C::Publish | C::Will)),
138            ResponseTopic(_) => (0x08, matches!(context, C::Publish | C::Will)),
139            CorrelationData(_) => (0x09, matches!(context, C::Publish | C::Will)),
140            SubscriptionIdentifier(_) => (0x0b, matches!(context, C::Publish | C::Subscribe)),
141            SessionExpiryInterval(_) => (
142                0x11,
143                matches!(context, C::Connect | C::ConnAck | C::Disconnect),
144            ),
145            AssignedClientIdentifier(_) => (0x12, context == C::ConnAck),
146            ServerKeepAlive(_) => (0x13, context == C::ConnAck),
147            AuthenticationMethod(_) => (0x15, matches!(context, C::Connect | C::ConnAck | C::Auth)),
148            AuthenticationData(_) => (0x16, matches!(context, C::Connect | C::ConnAck | C::Auth)),
149            RequestProblemInformation(_) => (0x17, context == C::Connect),
150            WillDelayInterval(_) => (0x18, context == C::Will),
151            RequestResponseInformation(_) => (0x19, context == C::Connect),
152            ResponseInformation(_) => (0x1a, context == C::ConnAck),
153            ServerReference(_) => (0x1c, matches!(context, C::ConnAck | C::Disconnect)),
154            ReasonString(_) => (
155                0x1f,
156                matches!(context, C::ConnAck | C::Ack | C::Disconnect | C::Auth),
157            ),
158            ReceiveMaximum(_) => (0x21, matches!(context, C::Connect | C::ConnAck)),
159            TopicAliasMaximum(_) => (0x22, matches!(context, C::Connect | C::ConnAck)),
160            TopicAlias(_) => (0x23, context == C::Publish),
161            MaximumQoS(_) => (0x24, context == C::ConnAck),
162            RetainAvailable(_) => (0x25, context == C::ConnAck),
163            UserProperty(_, _) => (0x26, true),
164            MaximumPacketSize(_) => (0x27, matches!(context, C::Connect | C::ConnAck)),
165            WildcardSubscriptionAvailable(_) => (0x28, context == C::ConnAck),
166            SubscriptionIdentifierAvailable(_) => (0x29, context == C::ConnAck),
167            SharedSubscriptionAvailable(_) => (0x2a, context == C::ConnAck),
168        };
169        require(allowed, "Property is not allowed in this packet")?;
170        let repeatable = id == 0x26 || (id == 0x0b && context == C::Publish);
171        require(
172            repeatable || seen & (1 << id) == 0,
173            "Duplicate singleton property",
174        )?;
175        seen |= 1 << id;
176        property_value(property)?;
177    }
178    if context == C::Connect {
179        require(
180            seen & (1 << 0x16) == 0 || seen & (1 << 0x15) != 0,
181            "CONNECT Authentication Data requires Authentication Method",
182        )?;
183    }
184    Ok(())
185}

187pub(super) fn reason(kind: ControlPacketType, code: u8) -> Result<(), ParseError> {
188    use ControlPacketType::*;
189    let allowed: &[u8] = match kind {
190        CONNACK => &[
191            0, 0x80, 0x81, 0x82, 0x83, 0x84, 0x85, 0x86, 0x87, 0x88, 0x89, 0x8a, 0x8c, 0x90, 0x95,
192            0x97, 0x99, 0x9a, 0x9b, 0x9c, 0x9d, 0x9f,
193        ],
194        PUBACK | PUBREC => &[0, 0x10, 0x80, 0x83, 0x87, 0x90, 0x91, 0x97, 0x99],
195        PUBREL | PUBCOMP => &[0, 0x92],
196        SUBACK => &[
197            0, 1, 2, 0x80, 0x83, 0x87, 0x8f, 0x91, 0x97, 0x9e, 0xa1, 0xa2,
198        ],
199        UNSUBACK => &[0, 0x11, 0x80, 0x83, 0x87, 0x8f, 0x91],
200        DISCONNECT => &[
201            0, 4, 0x80, 0x81, 0x82, 0x83, 0x87, 0x89, 0x8b, 0x8d, 0x8e, 0x8f, 0x90, 0x93, 0x94,
202            0x95, 0x96, 0x97, 0x98, 0x99, 0x9a, 0x9b, 0x9c, 0x9d, 0x9e, 0x9f, 0xa0, 0xa1, 0xa2,
203        ],
204        AUTH => &[0, 0x18, 0x19],
205        _ => &[],
206    };
207    require(
208        allowed.contains(&code),
209        "Invalid reason code for packet type",
210    )
211}

213pub(super) fn subscription(
214    subscription: &super::mqttv5::subscribev5::TopicSubscription,
215) -> Result<(), ParseError> {
216    super::validate_topic_filter(&subscription.topic_filter)?;
217    super::validate_shared_subscription(&subscription.topic_filter)?;
218    require(subscription.qos <= 2, "Subscription QoS cannot exceed 2")?;
219    require(
220        subscription.retain_handling <= 2,
221        "Retain Handling cannot exceed 2",
222    )?;
223    require(
224        !subscription.no_local || !subscription.topic_filter.starts_with("$share/"),
225        "No Local is not allowed on a shared subscription",
226    )
227}
