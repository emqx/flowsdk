14struct Subscription {
15    options: TopicSubscription,
16    identifier: Option<u32>,
17}

19/// In-memory MQTT 5 session with non-shared subscriptions and a local retained store.
20/// The caller supplies validated packets and drives acknowledgments/transmission.
21/// Broker-wide routing, shared subscriptions and persistent storage are external.
22/// Each matching subscription produces a separate publication with its granted
23/// QoS and optional Subscription Identifier, including retained replay.
24pub struct ServerSession {
25    // The client's subscriptions. The key is the topic filter.
26    subscriptions: HashMap<String, Subscription>,
27
28    // QoS 1 and QoS 2 messages that have been sent to the client but not acknowledged.
29    // The key is the packet identifier.
30    unacknowledged_publishes: HashMap<u16, MqttPublish>,
31
32    // Application messages awaiting their first transmission to the client.
33    pending_publishes: VecDeque<MqttPublish>,
34
35    // Incoming QoS 2 identifiers remain owned until PUBREL, independently of
36    // outgoing identifiers used when forwarding the application message.
37    received_qos2: HashSet<u16>,
38
39    retained_publishes: HashMap<String, MqttPublish>,
40    packet_id_counter: u16,
41
42    // QoS 2 PUBREL messages that have been sent but not yet acknowledged with PUBCOMP.
43    // The key is the packet identifier.
44    unacknowledged_pubrels: HashMap<u16, MqttPubRel>,
45
46    // The Will Message.
47    #[allow(dead_code)]
48    will: Option<MqttPublish>,
49
50    // The session expiry interval.
51    #[allow(dead_code)]
52    session_expiry_interval: u32,
53
54    // The client's receive maximum value.
55    receive_maximum: u16,
56}
57
58impl ServerSession {
59    pub fn new(receive_maximum: u16) -> Self {
60        ServerSession {
61            subscriptions: HashMap::new(),
62            unacknowledged_publishes: HashMap::new(),
63            pending_publishes: VecDeque::new(),
64            received_qos2: HashSet::new(),
65            retained_publishes: HashMap::new(),
66            packet_id_counter: 0,
67            unacknowledged_pubrels: HashMap::new(),
68            will: None,
69            session_expiry_interval: 0,
70            receive_maximum,
71        }
72    }

74    pub fn handle_incoming_subscribe(&mut self, subscribe: MqttSubscribe) {
75        // A SUBSCRIBE's identifier applies to all of its topic filters. Keeping
76        // it with the options also makes replacement and unsubscribe atomic.
77        let identifier = subscribe
78            .properties
79            .iter()
80            .find_map(|property| match property {
81                Property::SubscriptionIdentifier(identifier) => Some(*identifier),
82                _ => None,
83            });
84        for options in subscribe.subscriptions {
85            let subscription = Subscription {
86                options,
87                identifier,
88            };
89            let options = &subscription.options;
90            let existed = self.subscriptions.contains_key(&options.topic_filter);
91            if !options.topic_filter.starts_with("$share/")
92                && (options.retain_handling == 0 || (options.retain_handling == 1 && !existed))
93            {
94                for publish in self.retained_publishes.values() {
95                    if topic_matches(&options.topic_filter, &publish.topic_name) {
96                        self.pending_publishes.push_back(forwarded_publish(
97                            publish,
98                            &subscription,
99                            true,
100                        ));
101                    }
102                }
103            }
104            self.subscriptions
105                .insert(options.topic_filter.clone(), subscription);
106        }
107    }
108
109    pub fn handle_incoming_unsubscribe(&mut self, unsubscribe: MqttUnsubscribe) {
110        for topic_filter in unsubscribe.topic_filters {
111            self.subscriptions.remove(&topic_filter);
112        }
113    }

115    pub fn handle_incoming_publish(&mut self, publish: MqttPublish) -> Option<MqttPacket> {
116        if publish.qos == 2 && !self.received_qos2.insert(publish.packet_id.unwrap()) {
117            return Some(MqttPacket::PubRec5(MqttPubRec::new_success(
118                publish.packet_id.unwrap(),
119            )));
120        }
121        if publish.retain {
122            if publish.payload.is_empty() {
123                self.retained_publishes.remove(&publish.topic_name);
124            } else {
125                let mut retained = publish.clone();
126                retained.dup = false;
127                retained.packet_id = None;
128                self.retained_publishes
129                    .insert(publish.topic_name.clone(), retained);
130            }
131        }
132        for (filter, subscription) in &self.subscriptions {
133            if topic_matches(filter, &publish.topic_name) {
134                self.pending_publishes
135                    .push_back(forwarded_publish(&publish, subscription, false));
136            }
137        }
138
139        match publish.qos {
140            1 => Some(MqttPacket::PubAck5(
141                crate::mqtt_serde::mqttv5::puback::MqttPubAck {
142                    packet_id: publish.packet_id.unwrap(),
143                    reason_code: 0,
144                    properties: Vec::new(),
145                },
146            )),
147            2 => {
148                let pubrec = MqttPubRec {
149                    packet_id: publish.packet_id.unwrap(),
150                    reason_code: 0,
151                    properties: Vec::new(),
152                };
153                Some(MqttPacket::PubRec5(pubrec))
154            }
155            _ => None,
156        }
157    }
158
159    pub fn handle_incoming_puback(&mut self, puback: MqttPubAck) {
160        if self
161            .unacknowledged_publishes
162            .get(&puback.packet_id)
163            .is_some_and(|publish| publish.qos == 1)
164        {
165            self.unacknowledged_publishes.remove(&puback.packet_id);
166        }
167    }
168
169    pub fn handle_incoming_pubrec(&mut self, pubrec: MqttPubRec) -> Option<MqttPubRel> {
170        if let Some(pubrel) = self.unacknowledged_pubrels.get(&pubrec.packet_id) {
171            return (pubrec.reason_code < 0x80).then(|| pubrel.clone());
172        }
173        if self
174            .unacknowledged_publishes
175            .get(&pubrec.packet_id)
176            .is_none_or(|publish| publish.qos != 2)
177        {
178            return None;
179        }
180        self.unacknowledged_publishes.remove(&pubrec.packet_id);
181        if pubrec.reason_code < 0x80 {
182            let pubrel = MqttPubRel {
183                packet_id: pubrec.packet_id,
184                reason_code: 0,
185                properties: Vec::new(),
186            };
187            self.unacknowledged_pubrels
188                .insert(pubrec.packet_id, pubrel.clone());
189            Some(pubrel)
190        } else {
191            None
192        }
193    }
194
195    pub fn handle_incoming_pubrel(&mut self, pubrel: MqttPubRel) -> MqttPubComp {
196        let known = self.received_qos2.remove(&pubrel.packet_id);
197        MqttPubComp {
198            packet_id: pubrel.packet_id,
199            reason_code: if known { 0 } else { 0x92 },
200            properties: Vec::new(),
201        }
202    }
203
204    pub fn handle_incoming_pubcomp(&mut self, pubcomp: MqttPubComp) {
205        self.unacknowledged_pubrels.remove(&pubcomp.packet_id);
206    }

208    /// Explicitly resend outstanding exchanges and transmit queued publications.
209    /// MQTT 5 callers should request retransmission only when resuming a session.
210    /// Existing exchanges keep their quota reservation; PUBREL needs no new slot.
211    pub fn resend_pending_messages(&mut self) -> Vec<MqttPacket> {
212        let mut packets_to_resend = Vec::new();
213        for publish in self.unacknowledged_publishes.values() {
214            let mut publish = publish.clone();
215            publish.dup = true;
216            packets_to_resend.push(MqttPacket::Publish5(publish));
217        }
218
219        // Resend unacknowledged pubrels
220        for pubrel in self.unacknowledged_pubrels.values() {
221            packets_to_resend.push(MqttPacket::PubRel5(pubrel.clone()));
222        }
223        packets_to_resend.extend(self.take_pending_messages());
224        packets_to_resend
225    }
226
227    /// Drain new publications allowed by the send quota without retransmitting.
228    /// Call after receiving application messages, subscriptions or acknowledgments.
229    pub fn take_pending_messages(&mut self) -> Vec<MqttPacket> {
230        let mut packets = Vec::new();
231        let inflight = self.unacknowledged_publishes.len() + self.unacknowledged_pubrels.len();
232        let mut available_slots = (self.receive_maximum as usize).saturating_sub(inflight);
233
234        // Send pending publishes
235        for _ in 0..self.pending_publishes.len() {
236            let mut publish = self.pending_publishes.pop_front().unwrap();
237            if publish.qos > 0 {
238                if available_slots == 0 {
239                    self.pending_publishes.push_back(publish);
240                    continue;
241                }
242                let Some(id) = self.next_packet_id() else {
243                    self.pending_publishes.push_front(publish);
244                    break;
245                };
246                publish.packet_id = Some(id);
247                self.unacknowledged_publishes.insert(id, publish.clone());
248                available_slots -= 1;
249            }
250            packets.push(MqttPacket::Publish5(publish));
251        }
252        packets
253    }
254
255    fn next_packet_id(&mut self) -> Option<u16> {
256        for _ in 0..u16::MAX {
257            self.packet_id_counter = self.packet_id_counter.checked_add(1).unwrap_or(1);
258            if !self
259                .unacknowledged_publishes
260                .contains_key(&self.packet_id_counter)
261                && !self
262                    .unacknowledged_pubrels
263                    .contains_key(&self.packet_id_counter)
264            {
265                return Some(self.packet_id_counter);
266            }
267        }
268        None
269    }
270}

272fn forwarded_publish(
273    publish: &MqttPublish,
274    subscription: &Subscription,
275    retained_replay: bool,
276) -> MqttPublish {
277    let mut forwarded = publish.clone();
278    forwarded.qos = publish.qos.min(subscription.options.qos);
279    forwarded.dup = false;
280    forwarded.retain =
281        retained_replay || (publish.retain && subscription.options.retain_as_published);
282    // Identifiers belong to this receiving subscription, not the source message.
283    forwarded
284        .properties
285        .retain(|property| !matches!(property, Property::SubscriptionIdentifier(_)));
286    if let Some(identifier) = subscription.identifier {
287        forwarded
288            .properties
289            .push(Property::SubscriptionIdentifier(identifier));
290    }
291    // The subscriber's packet identifier is assigned when its send quota permits.
292    forwarded.packet_id = None;
293    forwarded
294}

296fn topic_matches(filter: &str, topic: &str) -> bool {
297    if topic.starts_with('$') && filter.starts_with(['+', '#']) {
298        return false;
299    }
300    let mut filters = filter.split('/');
301    let mut topics = topic.split('/');
302    loop {
303        match filters.next() {
304            Some("#") => return filters.next().is_none(),
305            Some("+") => {
306                if topics.next().is_none() {
307                    return false;
308                }
309            }
310            Some(level) if topics.next() != Some(level) => return false,
311            Some(_) => {}
312            None => return topics.next().is_none(),
313        }
314    }
