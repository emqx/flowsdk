15/// An outstanding MQTT exchange. Desktop callers retain `Instant` timestamps.
16#[derive(Debug, Clone)]
17pub struct InflightEntry<T: TimePoint = DefaultTime> {
18    pub packet_id: u16,
19    pub packet: MqttPacket,
20    pub sent_at: T,
21    pub retry_count: u32,
22    pub qos: u8,
23    /// The originating logical stream, or None for an ordinary byte transport.
24    pub stream: Option<u64>,
25}
26
27/// Inflight exchanges with receive-maximum accounting and ordered retransmission.
28///
29/// Lookups use a hash map on hosts and an O(log n) tree without `std`.
30/// Replay order is admission order, independent of map order or equal timestamps.
31/// MQTT 5 exchanges are replayed on reconnect, not on a retransmission timer.
32pub struct InflightQueue<T: TimePoint = DefaultTime> {
33    entries: Map<u16, InflightEntry<T>>,
34    admission_order: VecDeque<u16>,
35    deadline_queue: VecDeque<(u16, T)>,
36    receive_maximum: u16,
37    mqtt_version: u8,
38    retransmission_timeout: Duration,
39    publish_quota_used: usize,
40    last_update: Option<T>,
41}
42
43impl<T: TimePoint> InflightQueue<T> {
44    pub fn new(receive_maximum: u16, mqtt_version: u8, retransmission_timeout: Duration) -> Self {
45        Self {
46            entries: Map::new(),
47            admission_order: VecDeque::new(),
48            deadline_queue: VecDeque::new(),
49            receive_maximum: if receive_maximum == 0 {
50                u16::MAX
51            } else {
52                receive_maximum
53            },
54            mqtt_version,
55            retransmission_timeout,
56            publish_quota_used: 0,
57            last_update: None,
58        }
59    }
60
61    fn check_time(&self, now: T) -> Result<(), MqttClientError> {
62        if self.last_update.is_some_and(|last| now < last)
63            || (self.mqtt_version != 5 && now.checked_add(self.retransmission_timeout).is_none())
64        {
65            return Err(MqttClientError::InvalidConfiguration {
66                field: "monotonic_time".into(),
67                reason: "inflight timestamp moved backwards or its deadline overflowed".into(),
68            });
69        }
70        Ok(())
71    }
72
73    pub fn can_push_publish(&self) -> bool {
74        self.publish_quota_used < self.receive_maximum as usize
75    }
76
77    pub fn update_receive_maximum(&mut self, receive_maximum: u16) {
78        if receive_maximum > 0 {
79            self.receive_maximum = receive_maximum;
80        }
81    }
82
83    pub fn push_at(
84        &mut self,
85        packet_id: u16,
86        packet: MqttPacket,
87        qos: u8,
88        now: T,
89    ) -> Result<(), MqttClientError> {
90        self.push_with_stream_at(packet_id, packet, qos, None, now)
91    }
92
93    pub fn push_with_stream_at(
94        &mut self,
95        packet_id: u16,
96        packet: MqttPacket,
97        qos: u8,
98        stream: Option<u64>,
99        now: T,
100    ) -> Result<(), MqttClientError> {
101        self.check_time(now)?;
102        if packet_id == 0 || self.entries.contains_key(&packet_id) {
103            return Err(MqttClientError::InvalidPacketId { packet_id });
104        }
105        let publish = matches!(packet, MqttPacket::Publish5(_) | MqttPacket::Publish3(_));
106        if publish && !self.can_push_publish() {
107            return Err(MqttClientError::BufferFull {
108                buffer_type: "inflight_publish".to_string(),
109                capacity: self.receive_maximum as usize,
110            });
111        }
112        if publish {
113            self.publish_quota_used += 1;
114        }
115        self.entries.insert(
116            packet_id,
117            InflightEntry {
118                packet_id,
119                packet,
120                sent_at: now,
121                retry_count: 0,
122                qos,
123                stream,
124            },
125        );
126        self.admission_order.push_back(packet_id);
127        self.last_update = Some(now);
128        self.refresh_deadline(packet_id, now);
129        Ok(())
130    }
131
132    fn refresh_deadline(&mut self, packet_id: u16, now: T) {
133        if self.mqtt_version != 5 {
134            // Keep one live deadline per exchange, including PUBREL/resume.
135            self.deadline_queue.retain(|(id, _)| *id != packet_id);
136            let deadline = now
137                .checked_add(self.retransmission_timeout)
138                .expect("validated inflight deadline");
139            self.deadline_queue.push_back((packet_id, deadline));
140        }
141    }
142
143    pub fn get(&self, packet_id: u16) -> Option<&InflightEntry<T>> {
144        self.entries.get(&packet_id)
145    }
146    pub fn contains(&self, packet_id: u16) -> bool {
147        self.entries.contains_key(&packet_id)
148    }
149
150    pub fn transition_pubrel_at(
151        &mut self,
152        packet_id: u16,
153        packet: MqttPacket,
154        now: T,
155    ) -> Result<(), MqttClientError> {
156        self.check_time(now)?;
157        if let Some(entry) = self.entries.get_mut(&packet_id) {
158            entry.packet = packet;
159            entry.sent_at = now;
160            self.refresh_deadline(packet_id, now);
161        }
162        self.last_update = Some(now);
163        Ok(())
164    }
165
166    pub fn resume_on_stream_at(
167        &mut self,
168        packet_id: u16,
169        stream: u64,
170        now: T,
171    ) -> Result<(), MqttClientError> {
172        self.check_time(now)?;
173        if let Some(entry) = self.entries.get_mut(&packet_id) {
174            entry.stream = Some(stream);
175            entry.sent_at = now;
176            self.refresh_deadline(packet_id, now);
177        }
178        self.last_update = Some(now);
179        Ok(())
180    }
181
182    pub fn acknowledge(&mut self, packet_id: u16) -> Option<InflightEntry<T>> {
183        let entry = self.entries.remove(&packet_id)?;
184        self.admission_order.retain(|id| *id != packet_id);
185        self.deadline_queue.retain(|(id, _)| *id != packet_id);
186        if matches!(
187            entry.packet,
188            MqttPacket::Publish5(_)
189                | MqttPacket::Publish3(_)
190                | MqttPacket::PubRel5(_)
191                | MqttPacket::PubRel3(_)
192        ) {
193            self.publish_quota_used = self.publish_quota_used.saturating_sub(1);
194        }
195        Some(entry)
196    }
197
198    pub fn get_expired_at(&mut self, now: T) -> Result<Vec<MqttPacket>, MqttClientError> {
199        Ok(self
200            .get_expired_with_stream_at(now)?
201            .into_iter()
202            .map(|(packet, _)| packet)
203            .collect())
204    }
205
206    pub fn get_expired_with_stream_at(
207        &mut self,
208        now: T,
209    ) -> Result<Vec<(MqttPacket, Option<u64>)>, MqttClientError> {
210        self.check_time(now)?;
211        self.last_update = Some(now);
212        let mut expired = Vec::new();
213        // Limit to the original entries: even a zero timeout retries once per tick.
214        for _ in 0..self.deadline_queue.len() {
215            let Some(&(id, deadline)) = self.deadline_queue.front() else {
216                break;
217            };
218            if now < deadline {
219                break;
220            }
221            self.deadline_queue.pop_front();
222            if let Some(entry) = self.entries.get_mut(&id) {
223                entry.retry_count = entry.retry_count.saturating_add(1);
224                entry.sent_at = now;
225                expired.push((entry.packet.clone(), entry.stream));
226                self.refresh_deadline(id, now);
227            }
228        }
229        Ok(expired)
230    }
231
232    pub fn get_all_for_reconnect(&self) -> Vec<MqttPacket> {
233        self.admission_order
234            .iter()
235            .filter_map(|id| self.entries.get(id))
236            .map(|entry| entry.packet.clone())
237            .collect()
238    }
239
240    pub fn snapshot_for_reconnect(&self) -> Vec<(u16, MqttPacket)> {
241        self.admission_order
242            .iter()
243            .filter_map(|id| self.entries.get(id))
244            .map(|entry| (entry.packet_id, entry.packet.clone()))
245            .collect()
246    }
247
248    pub fn next_expiration(&self) -> Option<T> {
249        self.deadline_queue.front().map(|&(_, deadline)| deadline)
250    }
251
252    pub fn clear(&mut self) {
253        self.entries.clear();
254        self.admission_order.clear();
255        self.deadline_queue.clear();
256        self.publish_quota_used = 0;
257    }
258
259    pub fn len(&self) -> usize {
260        self.entries.len()
261    }
262    pub fn is_empty(&self) -> bool {
263        self.entries.is_empty()
264    }
265}

285    #[test]
286    fn test_inflight_push_ack() {
287        let mut q = InflightQueue::new(10, 5, Duration::from_secs(5));
288        assert!(q.can_push_publish());
289        q.push(1, create_packet(1), 1).unwrap();
290        assert_eq!(q.len(), 1);
291        q.acknowledge(1).unwrap();
292        assert_eq!(q.len(), 0);
293    }

295    #[test]
296    fn test_inflight_receive_maximum() {
297        let mut q = InflightQueue::new(2, 5, Duration::from_secs(5));
298        q.push(1, create_packet(1), 1).unwrap();
299        q.push(2, create_packet(2), 1).unwrap();
300        assert!(!q.can_push_publish());
301        assert!(q.push(3, create_packet(3), 1).is_err());
302
303        // Update limit
304        q.update_receive_maximum(5);
305        assert!(q.can_push_publish());
306        q.push(3, create_packet(3), 1).unwrap();
307        assert_eq!(q.len(), 3);
308    }

311    fn test_inflight_v3_retransmission() {
312        let mut q = InflightQueue::new(10, 3, Duration::from_secs(1));
313        let start = Instant::now();
314        q.push(1, create_packet(1), 1).unwrap();

316        let expired = q.get_expired(start + Duration::from_secs(2));
317        assert_eq!(expired.len(), 1);

321        let expired2 = q.get_expired(start + Duration::from_secs(4));
322        assert_eq!(expired2.len(), 1);

335    #[test]
336    fn test_inflight_v5_deadline_queue_cleanup() {
337        let mut q = InflightQueue::new(100, 5, Duration::from_secs(5));
338
339        // For MQTT v5, deadline_queue should never be populated
340        // Push and acknowledge many messages to simulate sustained load
341        for i in 1..=50 {
342            q.push(i, create_packet(i), 1).unwrap();
343        }
344
345        // Verify deadline_queue remains empty (never populated for v5)
346        assert_eq!(q.deadline_queue.len(), 0);
347        assert_eq!(q.len(), 50);
348
349        // Acknowledge all messages
350        for i in 1..=50 {
351            q.acknowledge(i);
352        }
353
354        // After acknowledging all, entries should be empty
355        assert_eq!(q.len(), 0);
356        // deadline_queue should still be empty
357        assert_eq!(q.deadline_queue.len(), 0);
358
359        // Push and acknowledge more messages to verify it stays empty
360        for i in 51..=100 {
361            q.push(i, create_packet(i), 1).unwrap();
362        }
363        assert_eq!(q.deadline_queue.len(), 0);
364        assert_eq!(q.len(), 50);
365
366        for i in 51..=100 {
367            q.acknowledge(i);
368        }
369        assert_eq!(q.len(), 0);
370        assert_eq!(q.deadline_queue.len(), 0);
371
372        // Verify get_expired() still works (returns empty for v5)
373        for i in 101..=110 {
374            q.push(i, create_packet(i), 1).unwrap();
375        }
376
377        let expired = q.get_expired(Instant::now());
378        assert!(expired.is_empty()); // MQTT v5 doesn't retransmit
379        assert_eq!(q.deadline_queue.len(), 0); // Still empty
380        assert_eq!(q.len(), 10); // Unacknowledged entries remain
381    }
