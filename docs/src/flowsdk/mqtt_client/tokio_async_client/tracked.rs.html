2//! Opt-in operation receipts. Enqueueing transfers publication ownership to the worker.
3use super::*;
4use std::sync::{Arc, OnceLock};
5use tokio::sync::Notify;
6use tokio::time::Instant as Deadline;
7
8#[derive(Debug)]
9pub struct TrackedPublishError {
10    /// Whether engine admission has been observed when this error is returned.
11    /// On timeout, `false` does not prevent later admission or make retrying safe.
12    pub admitted: bool,
13    pub error: MqttClientError,
14}
15
16#[derive(Debug)]
17pub(super) struct Completion {
18    at: Deadline,
19    result: Result<PublishResult, MqttClientError>,
20}
21
22#[derive(Debug)]
23pub(super) struct Receipt {
24    // Only the worker publishes these immutable milestones. Waiters never
25    // change the request or contend with admission over mutable receipt state.
26    pub generation: OnceLock<Arc<()>>,
27    pub completion: OnceLock<Completion>,
28    pub written: OnceLock<Result<(), MqttClientError>>,
29    changed: Notify,
30    pub deadline: Deadline,
31}
32
33impl Receipt {
34    pub fn new(deadline: Deadline) -> Self {
35        Self {
36            generation: OnceLock::new(),
37            completion: OnceLock::new(),
38            written: OnceLock::new(),
39            changed: Notify::new(),
40            deadline,
41        }
42    }
43
44    pub fn is_admitted(&self) -> bool {
45        self.generation.get().is_some()
46    }
47
48    pub fn fail(&self, error: MqttClientError) {
49        self.completion.get_or_init(|| Completion {
50            at: Deadline::now(),
51            result: Err(error.clone()),
52        });
53        self.written.get_or_init(|| Err(error));
54        self.changed.notify_waiters();
55    }
56
57    pub fn written(&self, result: Result<(), MqttClientError>) {
58        if self.written.set(result).is_ok() {
59            self.changed.notify_waiters();
60        }
61    }
62
63    pub fn complete(&self, result: Result<PublishResult, MqttClientError>) {
64        // Preserve confirmation timing even if nobody is currently waiting.
65        if self
66            .completion
67            .set(Completion {
68                at: Deadline::now(),
69                result,
70            })
71            .is_ok()
72        {
73            self.changed.notify_waiters();
74        }
75    }
76}

78/// One publication, independently identified from its reusable MQTT packet ID.
79/// Dropping this handle or cancelling `wait` never retracts a queued publication.
80pub struct PublishOperation {
81    pub(super) receipt: Arc<Receipt>,
82    timed_out: bool,
83}
84
85impl PublishOperation {
86    pub(super) fn new(receipt: Arc<Receipt>) -> Self {
87        Self {
88            receipt,
89            timed_out: false,
90        }
91    }
92
93    pub fn drain_receipt(&self) -> PublishDrain {
94        PublishDrain(self.receipt.clone())
95    }
96
97    /// QoS 0 completes at engine admission; QoS 1/2 at PUBACK/PUBCOMP (or rejection).
98    /// Timeout stops only this wait; even a command awaiting admission may still
99    /// publish afterward. A timeout is not evidence that retrying is safe.
100    /// Cancelling this wait preserves its original deadline. Confirmations recorded
101    /// before that deadline remain available when the wait resumes afterward.
102    pub async fn wait(&mut self) -> Result<PublishResult, TrackedPublishError> {
103        if self.timed_out {
104            return Err(self.timeout_error());
105        }
106        loop {
107            if let Some(result) = self.completed_result() {
108                return result;
109            }
110            if Deadline::now() >= self.receipt.deadline {
111                // A timely completion may have arrived between the first read
112                // and observing the deadline. Prefer it over a local timeout.
113                if let Some(result) = self.completed_result() {
114                    return result;
115                }
116                // Cache timeout only on this handle. It neither changes worker
117                // state nor poisons the independently observable write fence.
118                self.timed_out = true;
119                return Err(self.timeout_error());
120            }
121            // Register then recheck to avoid missing a concurrent completion.
122            // Already completed receipts never need a notification waiter.
123            let changed = self.receipt.changed.notified();
124            tokio::pin!(changed);
125            changed.as_mut().enable();
126            if self.receipt.completion.get().is_some() {
127                continue;
128            }
129            let _ = tokio::time::timeout_at(self.receipt.deadline, changed).await;
130        }
131    }
132
133    fn completed_result(&self) -> Option<Result<PublishResult, TrackedPublishError>> {
134        self.receipt.completion.get().map(|completion| {
135            if completion.at < self.receipt.deadline {
136                completion
137                    .result
138                    .clone()
139                    .map_err(|error| TrackedPublishError {
140                        admitted: self.receipt.is_admitted(),
141                        error,
142                    })
143            } else {
144                Err(self.timeout_error())
145            }
146        })
147    }
148
149    fn timeout_error(&self) -> TrackedPublishError {
150        TrackedPublishError {
151            admitted: self.receipt.is_admitted(),
152            error: MqttClientError::OperationTimeout {
153                operation: "tracked publish".into(),
154                timeout_ms: 0,
155            },
156        }
157    }
158}

160/// A per-operation local transport write-and-flush fence.
161/// Success does not prove broker delivery.
162#[derive(Clone)]
163pub struct PublishDrain(Arc<Receipt>);
164
165impl PublishDrain {
166    /// Compare the transports on which the operations were admitted. Every new
167    /// transport, including an intentional reconnect, has a distinct generation.
168    pub fn same_connection(&self, other: &Self) -> bool {
169        match (self.0.generation.get(), other.0.generation.get()) {
170            (Some(a), Some(b)) => Arc::ptr_eq(a, b),
171            _ => false,
172        }
173    }
174    pub fn is_written(&self) -> bool {
175        matches!(self.0.written.get(), Some(Ok(())))
176    }
177
178    /// Wait only through this operation, without disconnecting the shared client.
179    pub async fn drain_through(&self, deadline: Deadline) -> Result<(), MqttClientError> {
180        loop {
181            if let Some(result) = self.0.written.get() {
182                return result.clone();
183            }
184            let changed = self.0.changed.notified();
185            tokio::pin!(changed);
186            changed.as_mut().enable();
187            if self.0.written.get().is_some() {
188                continue;
189            }
190            tokio::time::timeout_at(deadline, changed)
191                .await
192                .map_err(|_| MqttClientError::OperationTimeout {
193                    operation: "publish drain".into(),
194                    timeout_ms: 0,
195                })?;
196        }
197    }
198}

200#[derive(Debug)]
201pub(super) struct TrackedCommand {
202    pub command: PublishCommand,
203    pub receipt: Arc<Receipt>,
204}
205
206impl Drop for TrackedCommand {
207    fn drop(&mut self) {
208        // Also wakes waiters when the worker is aborted with queued commands.
209        if !self.receipt.is_admitted() {
210            self.receipt.fail(MqttClientError::OperationCancelled {
211                operation: "publish worker stopped before admission".into(),
212            });
213        }
214    }
215}

217impl TokioAsyncMqttClient {
218    /// Enqueue without yielding, so the caller can retain its receipt atomically.
219    /// A full command channel has not admitted the publication and is retryable.
220    /// After successful enqueueing, dropping the operation or reaching `deadline`
221    /// does not cancel publication. The deadline bounds confirmation waiting only.
222    pub fn try_publish_tracked(
223        &self,
224        command: PublishCommand,
225        deadline: Deadline,
226    ) -> Result<PublishOperation, MqttClientError> {
227        let receipt = Arc::new(Receipt::new(deadline));
228        self.command_tx
229            .try_send(TokioClientCommand::PublishTracked(TrackedCommand {
230                command,
231                receipt: receipt.clone(),
232            }))
233            .map_err(|error| match error {
234                mpsc::error::TrySendError::Full(_) => MqttClientError::BufferFull {
235                    buffer_type: "tracked command queue".into(),
236                    capacity: self._config.command_queue_size,
237                },
238                mpsc::error::TrySendError::Closed(_) => MqttClientError::ChannelClosed {
239                    channel: "tracked command queue".into(),
240                },
241            })?;
242        Ok(PublishOperation::new(receipt))
243    }
