3//! Pure Sans-I/O MQTT Client
4//!
5//! This module provides a protocol-only MQTT client that is completely independent
6//! of any I/O runtime or networking implementation. It's designed for:
7//!
8//! - **Embedded systems** with custom I/O
9//! - **FFI bindings** (C, Python, Java, etc.)
10//! - **Custom runtimes** (WASM, bare metal, etc.)
11//! - **Testing** without network dependencies
12//! - **Protocol analysis** and debugging
13//!
14//! # Architecture
15//!
16//! The `NoIoMqttClient` is a thin wrapper around [`MqttEngine`] that provides
17//! a clean Sans-I/O interface. You are responsible for:
18//!
19//! 1. **Network I/O**: Reading/writing bytes to/from sockets
20//! 2. **Timers**: Calling `handle_tick()` at appropriate intervals
21//! 3. **Event Loop**: Coordinating between network, timers, and protocol
22//!
23//! # Portable usage
24//!
25//! ```
26//! use flowsdk::mqtt_client::{MqttClientOptions, PortableNoIoMqttClient};
27//! use flowsdk::time::Timestamp;
28//!
29//! let now = Timestamp::try_from_millis(12_345).unwrap();
30//! let mut client = PortableNoIoMqttClient::try_new_at(MqttClientOptions::default(), now).unwrap();
31//! client.connect_at(now).unwrap();
32//! let outgoing = client.take_outgoing_at(now).unwrap();
33//! assert_eq!(outgoing[0], 0x10);
34//! // Send outgoing bytes through an application-owned transport. Feed responses
35//! // through handle_incoming_at(bytes, now) and timers through handle_tick_at(now).
36//! ```
37//!
38//! Use one monotonic origin for every operation. Equal timestamps are valid;
39//! backward time and deadline overflow are rejected before changing state.
40//! The application supplies allocation, networking and scheduling. With `std`,
41//! the existing `NoIoMqttClient::new` and host-clock methods are also available.

56/// A pure Sans-I/O MQTT client.
57///
58/// This client handles all MQTT protocol logic without performing any I/O operations.
59/// You are responsible for:
60///
61/// - Reading bytes from the network and feeding them via [`handle_incoming_at()`](Self::handle_incoming_at)
62/// - Writing bytes from [`take_outgoing_at()`](Self::take_outgoing_at) to the network
63/// - Calling [`handle_tick_at()`](Self::handle_tick_at) at appropriate intervals for keep-alive and retransmissions
64///
65/// # Thread Safety
66///
67/// This client is `Send` and `Sync`; protocol operations require exclusive access.
68///
69/// # Example
70///
71/// ```no_run
72/// use flowsdk::mqtt_client::{PortableNoIoMqttClient as NoIoMqttClient, MqttClientOptions};
73///
74/// let options = MqttClientOptions::builder()
75///     .peer("broker.example.com:1883")
76///     .client_id("my_client")
77///     .build();
78///
79/// let mut client = NoIoMqttClient::new_at(options, flowsdk::time::Timestamp::ZERO);
80/// client.connect_at(flowsdk::time::Timestamp::ZERO).unwrap();
81///
82/// // Get CONNECT packet bytes
83/// let bytes = client.take_outgoing_at(flowsdk::time::Timestamp::ZERO).unwrap();
84/// // ... send `bytes` to your socket ...
85/// ```
86pub struct NoIoMqttClient<T: TimePoint = DefaultTime> {
87    engine: MqttEngine<T>,
88}

144    /// Get the next time when [`handle_tick_at()`](Self::handle_tick_at) should be called.
145    ///
146    /// Use this to optimize your event loop by only waking up when necessary.
147    /// Returns `None` if no timer is needed, including during disconnection.
148    ///
149    /// # Returns
150    ///
151    /// The next timestamp when a protocol timer needs to fire, or `None`.
152    ///
153    /// # Example
154    ///
155    /// ```no_run
156    /// # use flowsdk::mqtt_client::{PortableNoIoMqttClient as NoIoMqttClient, MqttClientOptions};
157    /// # use flowsdk::time::Timestamp;
158    /// # let client = NoIoMqttClient::new_at(MqttClientOptions::default(), flowsdk::time::Timestamp::ZERO);
159    /// if let Some(next_tick) = client.next_tick_at() {
160    ///     let now = Timestamp::ZERO;
161    ///     if next_tick > now {
162    ///         let sleep_duration = next_tick.saturating_duration_since(now);
163    ///         // Sleep for `sleep_duration` or until network data arrives
164    ///     }
165    /// }
166    /// ```
167    pub fn next_tick_at(&self) -> Option<T> {
168        self.engine.next_tick_at()
169    }

366    #[test]
367    fn test_create_client() {
368        let options = MqttClientOptions::builder()
369            .peer("localhost:1883")
370            .client_id("test_client")
371            .build();
372
373        let client = NoIoMqttClient::new(options);
374        assert!(!client.is_connected());
375        assert_eq!(client.mqtt_version(), 5); // Default is v5
376    }

378    #[test]
379    fn test_connect_generates_packet() {
380        let options = MqttClientOptions::builder()
381            .peer("localhost:1883")
382            .client_id("test_client")
383            .build();
384
385        let mut client = NoIoMqttClient::new(options);
386        client.connect().unwrap();
387
388        let outgoing = client.take_outgoing();
389        assert!(!outgoing.is_empty());
390        // CONNECT packet starts with 0x10 (packet type)
391        assert_eq!(outgoing[0] & 0xF0, 0x10);
392    }

394    #[test]
395    fn test_handle_connack() {
396        let options = MqttClientOptions::builder()
397            .peer("localhost:1883")
398            .client_id("test_client")
399            .build();
400
401        let mut client = NoIoMqttClient::new(options);
402        client.connect().unwrap();
403        let _ = client.take_outgoing();
404
405        // Simulate CONNACK for MQTT v5: 0x20 (type), 0x03 (length), 0x00 (flags), 0x00 (reason code), 0x00 (properties length)
406        let connack = vec![0x20, 0x03, 0x00, 0x00, 0x00];
407        let events = client.handle_incoming(&connack);
408
409        assert!(!events.is_empty(), "Should have received events");
410        assert!(
411            matches!(events[0], MqttEvent::Connected(_)),
412            "First event should be Connected, got: {:?}",
413            events[0]
414        );
415        assert!(client.is_connected());
416    }
417
418    #[test]
419    fn test_publish_qos0() {
420        let options = MqttClientOptions::builder()
421            .peer("localhost:1883")
422            .client_id("test_client")
423            .build();
424
425        let mut client = NoIoMqttClient::new(options);
426
427        // Connect first
428        client.connect().unwrap();
429        let _ = client.take_outgoing();
430
431        // Simulate CONNACK
432        let connack = vec![0x20, 0x03, 0x00, 0x00, 0x00];
433        let _ = client.handle_incoming(&connack);
434        assert!(client.is_connected());
435
436        // Now publish
437        let cmd = PublishCommand::simple("test/topic", b"payload".to_vec(), 0, false);
438        let result = client.publish(cmd);
439
440        // QoS 0 returns None for packet ID
441        assert!(result.is_ok());
442        assert_eq!(result.unwrap(), None);
443
444        let outgoing = client.take_outgoing();
445        assert!(!outgoing.is_empty());
446    }
