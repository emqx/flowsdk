91    pub(super) fn connect_properties(&self) -> Result<Vec<Property>, MqttClientError> {
92        use Property::*;
93        let mut properties = self.options.connect_properties.clone();
94        if let Some(v) = self.options.session_expiry_interval {
95            properties.push(SessionExpiryInterval(v));
96        }
97        if let Some(v) = self.options.maximum_packet_size {
98            properties.push(MaximumPacketSize(v));
99        }
100        if let Some(v) = self.options.request_response_information {
101            properties.push(RequestResponseInformation(v.into()));
102        }
103        if let Some(v) = self.options.request_problem_information {
104            properties.push(RequestProblemInformation(v.into()));
105        }
106        if let Some(v) = self.options.incoming_receive_maximum {
107            properties.push(ReceiveMaximum(v));
108        }
109        let mut merged = Vec::new();
110        for property in properties {
111            match &property {
112                ReceiveMaximum(0) | MaximumPacketSize(0) => {
113                    return Err(invalid(
114                        "CONNECT",
115                        "Receive Maximum and Maximum Packet Size must be nonzero",
116                    ))
117                }
118                RequestResponseInformation(v) | RequestProblemInformation(v) if *v > 1 => {
119                    return Err(invalid("CONNECT", "Boolean property must be 0 or 1"))
120                }
121                SessionExpiryInterval(_)
122                | ReceiveMaximum(_)
123                | MaximumPacketSize(_)
124                | TopicAliasMaximum(_)
125                | RequestResponseInformation(_)
126                | RequestProblemInformation(_)
127                | UserProperty(_, _)
128                | AuthenticationMethod(_)
129                | AuthenticationData(_) => {}
130                _ => return Err(invalid("CONNECT", "Property is not valid in CONNECT")),
131            }
132            if !matches!(property, UserProperty(_, _)) {
133                if let Some(existing) = merged
134                    .iter()
135                    .find(|p| std::mem::discriminant(*p) == std::mem::discriminant(&property))
136                {
137                    if existing != &property {
138                        return Err(invalid("CONNECT", "Conflicting singleton property values"));
139                    }
140                    continue;
141                }
142            }
143            merged.push(property);
144        }
145        if merged.iter().any(|p| matches!(p, AuthenticationData(_)))
146            && !merged.iter().any(|p| matches!(p, AuthenticationMethod(_)))
147        {
148            return Err(invalid(
149                "CONNECT",
150                "Authentication Data requires Authentication Method",
151            ));
152        }
153        Ok(merged)
154    }

156    pub(super) fn negotiate(&mut self, properties: &[Property]) -> Result<(), MqttClientError> {
157        use Property::*;
158        let mut seen = HashSet::new();
159        for p in properties {
160            if !matches!(p, UserProperty(_, _)) && !seen.insert(std::mem::discriminant(p)) {
161                return Err(invalid("CONNACK", "Duplicate singleton property"));
162            }
163            match p {
164                ReceiveMaximum(0) | MaximumPacketSize(0) => {
165                    return Err(invalid("CONNACK", "Limit must be nonzero"))
166                }
167                MaximumQoS(v) if *v > 1 => {
168                    return Err(invalid("CONNACK", "Maximum QoS must be 0 or 1"))
169                }
170                RetainAvailable(v)
171                | WildcardSubscriptionAvailable(v)
172                | SharedSubscriptionAvailable(v)
173                | SubscriptionIdentifierAvailable(v)
174                    if *v > 1 =>
175                {
176                    return Err(invalid("CONNACK", "Boolean property must be 0 or 1"))
177                }
178                ReceiveMaximum(v) => self
179                    .inflight_queue
180                    .update_receive_maximum((*v).min(self.options.receive_maximum)),
181                MaximumPacketSize(v) => self.reliability.maximum_packet_size = Some(*v as usize),
182                ServerKeepAlive(v) => self.reliability.keep_alive = *v,
183                MaximumQoS(v) => self.reliability.maximum_qos = *v,
184                RetainAvailable(v) => self.reliability.retain_available = *v != 0,
185                WildcardSubscriptionAvailable(v) => self.reliability.wildcard_available = *v != 0,
186                SharedSubscriptionAvailable(v) => self.reliability.shared_available = *v != 0,
187                SubscriptionIdentifierAvailable(v) => {
188                    self.reliability.subscription_ids_available = *v != 0
189                }
190                TopicAliasMaximum(v) => self.reliability.outgoing_alias_maximum = *v,
191                AssignedClientIdentifier(v) if !v.is_empty() => self.options.client_id = v.clone(),
192                SessionExpiryInterval(v) => self.reliability.session_expiry = *v,
193                AssignedClientIdentifier(_) => {
194                    return Err(invalid("CONNACK", "Assigned client ID is empty"))
195                }
196                AuthenticationMethod(_)
197                | AuthenticationData(_)
198                | ResponseInformation(_)
199                | ServerReference(_)
200                | ReasonString(_)
201                | UserProperty(_, _) => {}
202                _ => return Err(invalid("CONNACK", "Property is not valid in CONNACK")),
203            }
204        }
205        // The next CONNECT uses the assigned identifier, including any change
206        // in the encoded Remaining Length. Sizing must not alter negotiated state.
207        let reserve = self
208            .connect_packet()?
209            .to_bytes()
210            .map_err(MqttClientError::from)?
211            .len()
212            .max(6);
213        if self
214            .options
215            .max_outgoing_buffer_bytes
216            .is_some_and(|limit| reserve > limit)
217        {
218            return Err(invalid(
219                "max_outgoing_buffer_bytes",
220                "The reconnect CONNECT packet exceeds the outgoing byte limit",
221            ));
222        }
223        self.reliability.control_reserve = reserve;
224        Ok(())
225    }

227    pub(super) fn candidate_id(&mut self, requested: Option<u16>) -> Result<u16, MqttClientError> {
228        if let Some(id) = requested {
229            if id == 0
230                || self.reliability.reserved.contains(&id)
231                || self.inflight_queue.contains(id)
232            {
233                return Err(MqttClientError::InvalidPacketId { packet_id: id });
234            }
235            Ok(id)
236        } else {
237            self.next_packet_id()
238        }
239    }

241    pub(super) fn start_deadline(&mut self, operation: OperationKind, id: Option<u16>) {
242        let timeouts = self.options.operation_timeouts;
243        let duration = match operation {
244            OperationKind::Connect => timeouts.connect,
245            OperationKind::Publish => timeouts.publish,
246            OperationKind::Subscribe => timeouts.subscribe,
247            OperationKind::Unsubscribe => timeouts.unsubscribe,
248        };
249        if let Some(duration) = duration {
250            self.reliability
251                .deadlines
252                .insert((operation, id), (Instant::now() + duration, duration));
253        }
254    }

256    pub(super) fn finish_operation(&mut self, id: u16) {
257        self.reliability.reserved.remove(&id);
258        if self.reliability.reserved.is_empty() {
259            self.reliability.replay_reserve = 0;
260        }
261        self.reliability
262            .deadlines
263            .retain(|(_, pid), _| *pid != Some(id));
264    }

266    pub(super) fn check_deadlines(&mut self, now: Instant) {
267        let expired: Vec<_> = self
268            .reliability
269            .deadlines
270            .iter()
271            .filter(|(_, (at, _))| now >= *at)
272            .map(|(key, (_, duration))| (*key, *duration))
273            .collect();
274        for ((operation, packet_id), duration) in expired {
275            self.reliability.deadlines.remove(&(operation, packet_id));
276            self.events.push(MqttEvent::OperationFailed {
277                operation,
278                packet_id,
279                error: MqttClientError::OperationTimeout {
280                    operation: format!("{operation:?}"),
281                    timeout_ms: duration.as_millis().min(u64::MAX as u128) as u64,
282                },
283            });
284            if operation == OperationKind::Connect {
285                self.connection_lost_at(now);
286            }
287        }
288    }

290    pub(super) fn validate_publish(
291        &self,
292        command: &mut PublishCommand,
293    ) -> Result<Option<(u16, String)>, MqttClientError> {
294        if command.qos > self.reliability.maximum_qos
295            || (command.retain && !self.reliability.retain_available)
296        {
297            return Err(invalid(
298                "PUBLISH",
299                "Broker does not support requested QoS or retain",
300            ));
301        }
302        let aliases: Vec<_> = command
303            .properties
304            .iter()
305            .filter_map(|p| {
306                if let Property::TopicAlias(v) = p {
307                    Some(*v)
308                } else {
309                    None
310                }
311            })
312            .collect();
313        let mut registration = None;
314        if let Some(&alias) = aliases.first() {
315            if aliases.len() != 1 || alias == 0 || alias > self.reliability.outgoing_alias_maximum {
316                return Err(invalid(
317                    "Topic Alias",
318                    "Alias exceeds negotiated maximum or is duplicated",
319                ));
320            }
321            if command.topic_name.is_empty() {
322                command.topic_name = self
323                    .reliability
324                    .outgoing_aliases
325                    .get(&alias)
326                    .cloned()
327                    .ok_or_else(|| invalid("Topic Alias", "Undefined alias"))?;
328            } else {
329                registration = Some((alias, command.topic_name.clone()));
330            }
331        }
332        Ok(registration)
333    }

335    pub(super) fn validate_subscribe(
336        &self,
337        command: &SubscribeCommand,
338    ) -> Result<(), MqttClientError> {
339        if !self.reliability.subscription_ids_available
340            && command
341                .properties
342                .iter()
343                .any(|p| matches!(p, Property::SubscriptionIdentifier(_)))
344        {
345            return Err(invalid(
346                "SUBSCRIBE",
347                "Broker does not support subscription identifiers",
348            ));
349        }
350        for s in &command.subscriptions {
351            if (!self.reliability.wildcard_available && s.topic_filter.contains(['+', '#']))
352                || (!self.reliability.shared_available && s.topic_filter.starts_with("$share/"))
353            {
354                return Err(invalid(
355                    "SUBSCRIBE",
356                    "Broker does not support this subscription",
357                ));
358            }
359        }
360        Ok(())
361    }

364        &self,
365        packet: &MqttPacket,

367        let (qos, retain) = match packet {

373            return Err(invalid(
374                "PUBLISH",
375                "Broker does not support requested QoS or retain",
376            ));
377        }
378        Ok(())
379    }

381    pub(super) fn validate_input_buffer(
382        options: &MqttClientOptions,
383        bytes: &[u8],
384    ) -> Result<(), MqttClientError> {
385        if let Some(capacity) = options.max_incoming_buffer_bytes {
386            if bytes.len() > capacity {
387                return Err(MqttClientError::BufferFull {
388                    buffer_type: "incoming bytes".into(),
389                    capacity,
390                });
391            }
392        }
393        let advertised = options.maximum_packet_size.map(|v| v as usize).or_else(|| {
394            options.connect_properties.iter().find_map(|p| {
395                if let Property::MaximumPacketSize(v) = p {
396                    Some(*v as usize)
397                } else {
398                    None
399                }
400            })
401        });
402        if let Some(capacity) = options
403            .max_incoming_packet_size
404            .into_iter()
405            .chain(advertised)
406            .min()
407        {
408            if bytes.len() >= 2 {
409                if let Ok((length, n)) = crate::mqtt_serde::parser::parse_vbi(&bytes[1..]) {
410                    if length + n + 1 > capacity {
411                        return Err(MqttClientError::BufferFull {
412                            buffer_type: "incoming packet".into(),
413                            capacity,
414                        });
415                    }
416                }
417            }
418        }
419        Ok(())
420    }

422    pub(super) fn check_packet_size(&self, bytes: &[u8]) -> Result<(), MqttClientError> {
423        if let Some(capacity) = self.options.max_outgoing_buffer_bytes {
424            if bytes.len() > capacity {
425                return Err(MqttClientError::BufferFull {
426                    buffer_type: "outgoing packet".into(),
427                    capacity,
428                });
429            }
430        }
431        if self
432            .reliability
433            .maximum_packet_size
434            .is_some_and(|max| bytes.len() > max)
435        {
436            return Err(invalid(
437                "packet size",
438                "Packet exceeds the broker's Maximum Packet Size",
439            ));
440        }
441        Ok(())
442    }

460    pub(super) fn flush_responses(&mut self) {
461        while let Some(packet) = self.reliability.responses.front().cloned() {
462            if let Some(capacity) = self.options.max_outgoing_buffer_bytes {
463                if packet.to_bytes().is_ok_and(|bytes| bytes.len() > capacity) {
464                    self.fail_connection(MqttClientError::BufferFull {
465                        buffer_type: "protocol response".into(),
466                        capacity,
467                    });
468                    break;
469                }
470            }
471            match self.enqueue_packet(packet) {
472                Ok(()) => {
473                    self.reliability.responses.pop_front();
474                }
475                Err(MqttClientError::BufferFull { .. }) => break,
476                Err(error) => {
477                    self.fail_connection(error);
478                    break;
479                }
480            }
481        }
482    }

497    pub(super) fn accept_incoming(
498        &mut self,
499        packet_type: crate::mqtt_serde::control_packet::ControlPacketType,
500    ) -> bool {
501        use crate::mqtt_serde::control_packet::ControlPacketType::*;
502        if self.reliability.input_closed {
503            return false;
504        }
505        let valid = match packet_type {
506            CONNACK => self.reliability.connecting && !self.is_connected,
507            AUTH | DISCONNECT => {
508                self.is_connected || (self.mqtt_version() == 5 && self.reliability.connecting)
509            }
510            _ => self.is_connected,
511        };
512        if !valid {
513            self.fail_connection(MqttClientError::ProtocolViolation {
514                message: format!("Unexpected {packet_type:?} in the current connection state"),
515            });
516        }
517        valid
518    }

526    pub(super) fn resolve_incoming(
527        &mut self,
528        publish: &mut MqttPublish,
529    ) -> Result<(), MqttClientError> {
530        let aliases: Vec<_> = publish
531            .properties
532            .iter()
533            .filter_map(|p| {
534                if let Property::TopicAlias(v) = p {
535                    Some(*v)
536                } else {
537                    None
538                }
539            })
540            .collect();
541        if let Some(&alias) = aliases.first() {
542            if aliases.len() != 1 || alias == 0 || alias > self.reliability.incoming_alias_maximum {
543                return Err(invalid("Topic Alias", "Invalid incoming alias"));
544            }
545            if publish.topic_name.is_empty() {
546                publish.topic_name = self
547                    .reliability
548                    .incoming_aliases
549                    .get(&alias)
550                    .cloned()
551                    .ok_or_else(|| invalid("Topic Alias", "Undefined incoming alias"))?;
552            } else {
553                self.reliability
554                    .incoming_aliases
555                    .insert(alias, publish.topic_name.clone());
556            }
557        }
558        if publish.topic_name.is_empty() {
559            return Err(invalid("PUBLISH", "Empty topic without a defined alias"));
560        }
561        Ok(())
562    }

564    pub(super) fn receive_publish(
565        &mut self,
566        qos: u8,
567        id: Option<u16>,
568        stream: Option<u64>,
569    ) -> Result<bool, MqttClientError> {
570        let Some(id) = id.filter(|_| qos > 0) else {
571            return Ok(true);
572        };
573        if id == 0 {
574            return Err(MqttClientError::InvalidPacketId { packet_id: id });
575        }
576        if let Some((stage, previous_stream)) = self.reliability.received.get(&id) {
577            if (qos == 1) != matches!(stage, ReceiveStage::Publish(1)) {
578                return Err(MqttClientError::InvalidPacketId { packet_id: id });
579            }
580            if previous_stream.is_some() && *previous_stream != stream {
581                return Err(invalid("PUBLISH", "QoS handshake moved between streams"));
582            }
583        }
584        if !self.reliability.received_on_connection.contains(&id) {
585            if self.reliability.incoming_quota_used
586                >= self.reliability.incoming_receive_maximum as usize
587            {
588                return Err(MqttClientError::ProtocolViolation {
589                    message: "Incoming Receive Maximum exceeded".into(),
590                });
591            }
592            self.reliability.received_on_connection.insert(id);
593            self.reliability.incoming_quota_used += 1;
594        }
595        if let Some((_, previous_stream)) = self.reliability.received.get_mut(&id) {
596            *previous_stream = stream;
597            return Ok(false);
598        }
599        self.reliability
600            .received
601            .insert(id, (ReceiveStage::Publish(qos), stream));
602        Ok(true)
603    }

605    pub(super) fn manual_ack_packet(
606        &self,
607        id: u16,
608        kind: u8,
609        reason: u8,
610        properties: Vec<Property>,
611        stream: Option<u64>,
612    ) -> Result<MqttPacket, MqttClientError> {
613        if id == 0 {
614            return Err(MqttClientError::InvalidPacketId { packet_id: id });
615        }
616        let valid = self
617            .reliability
618            .received
619            .get(&id)
620            .is_some_and(|(stage, channel)| {
621                *channel == stream
622                    && matches!(
623                        (kind, stage),
624                        (4, ReceiveStage::Publish(1))
625                            | (5, ReceiveStage::Publish(2))
626                            | (5, ReceiveStage::PubRec)
627                            | (7, ReceiveStage::PubRel)
628                    )
629            });
630        if !valid {
631            return Err(MqttClientError::InvalidPacketId { packet_id: id });
632        }
633        if self.mqtt_version() != 5 && (reason != 0 || !properties.is_empty()) {
634            return Err(invalid(
635                "acknowledgement",
636                "MQTT 3 does not support reason codes or properties",
637            ));
638        }
639        if (kind == 7 && !matches!(reason, 0 | 0x92))
640            || (kind != 7
641                && !matches!(
642                    reason,
643                    0 | 0x10 | 0x80 | 0x83 | 0x87 | 0x90 | 0x91 | 0x97 | 0x99
644                ))
645        {
646            return Err(invalid("acknowledgement", "Invalid reason code"));
647        }
648        if properties
649            .iter()
650            .any(|p| !matches!(p, Property::ReasonString(_) | Property::UserProperty(_, _)))
651            || properties
652                .iter()
653                .filter(|p| matches!(p, Property::ReasonString(_)))
654                .count()
655                > 1
656        {
657            return Err(invalid("acknowledgement", "Invalid properties"));
658        }
659        Ok(match (self.mqtt_version(), kind) {
660            (5, 4) => MqttPacket::PubAck5(MqttPubAck::new(id, reason, properties)),
661            (5, 5) => MqttPacket::PubRec5(MqttPubRec::new(id, reason, properties)),
662            (5, 7) => MqttPacket::PubComp5(MqttPubComp::new(id, reason, properties)),
663            (_, 4) => MqttPacket::PubAck3(crate::mqtt_serde::mqttv3::pubackv3::MqttPubAck::new(id)),
664            (_, 5) => MqttPacket::PubRec3(crate::mqtt_serde::mqttv3::pubrecv3::MqttPubRec::new(id)),
665            (_, 7) => {
666                MqttPacket::PubComp3(crate::mqtt_serde::mqttv3::pubcompv3::MqttPubComp::new(id))
667            }
668            _ => unreachable!(),
669        })
670    }

672    pub(super) fn commit_receive_ack(&mut self, id: u16, kind: u8, reason: u8) {
673        if kind == 5 && reason < 0x80 {
674            if let Some((stage, _)) = self.reliability.received.get_mut(&id) {
675                *stage = ReceiveStage::PubRec;
676            }
677        } else {
678            self.reliability.received.remove(&id);
679            self.reliability.received_on_connection.remove(&id);
680            // Even PUBCOMP for a resumed PUBREL replenishes the peer's quota,
681            // capped at the initial limit (MQTT 5 section 4.9).
682            self.reliability.incoming_quota_used =
683                self.reliability.incoming_quota_used.saturating_sub(1);
684        }
685    }
686
687    fn manual_ack(
688        &mut self,
689        id: u16,
690        kind: u8,
691        reason: u8,
692        properties: Vec<Property>,
693    ) -> Result<(), MqttClientError> {
694        let packet = self.manual_ack_packet(id, kind, reason, properties, None)?;
695        self.enqueue_packet(packet)?;
696        self.commit_receive_ack(id, kind, reason);
697        Ok(())
698    }
699
700    pub fn puback(
701        &mut self,
702        id: u16,
703        reason: u8,
704        properties: Vec<Property>,
705    ) -> Result<(), MqttClientError> {
706        self.manual_ack(id, 4, reason, properties)
707    }

709        &mut self,
710        id: u16,
711        reason: u8,
712        properties: Vec<Property>,
713    ) -> Result<(), MqttClientError> {

716    pub fn pubcomp(
717        &mut self,
718        id: u16,

726impl MqttEngine {
727    pub(super) fn receive_message(
728        &mut self,
729        mut publish: MqttPublish,
730        stream: Option<u64>,
731    ) -> (Vec<MqttEvent>, Vec<MqttPacket>) {
732        if let Err(error) = self.resolve_incoming(&mut publish) {
733            self.fail_connection(error);
734            return (self.take_events(), vec![]);
735        }
736        let qos = publish.qos;
737        let pid = publish.packet_id;
738        let deliver = match self.receive_publish(qos, pid, stream) {
739            Ok(deliver) => deliver,
740            Err(error) => {
741                self.fail_connection(error);
742                return (self.take_events(), vec![]);
743            }
744        };
745        let mut events = Vec::new();
746        if deliver {
747            events.push(MqttEvent::PublishReceived {
748                packet_id: pid,
749                stream,
750            });
751            events.push(MqttEvent::MessageReceived(publish));
752        }
753        let mut responses = Vec::new();
754        if let Some(id) = pid.filter(|_| qos > 0) {
755            let already_acknowledged = self
756                .reliability
757                .received
758                .get(&id)
759                .is_some_and(|(stage, _)| *stage == ReceiveStage::PubRec);
760            if self.options.auto_ack || already_acknowledged {
761                let kind = if qos == 1 { 4 } else { 5 };
762                match self.manual_ack_packet(id, kind, 0, vec![], stream) {
763                    Ok(packet) => {
764                        responses.push(packet);
765                        self.commit_receive_ack(id, kind, 0);
766                    }
767                    Err(error) => events.push(MqttEvent::Error(error)),
768                }
769            }
770        }
771        (events, responses)
772    }
773
774    pub(super) fn receive_pubrel(
775        &mut self,
776        id: u16,
777        stream: Option<u64>,
778    ) -> (Vec<MqttEvent>, Vec<MqttPacket>) {
779        if id == 0 {
780            self.fail_connection(MqttClientError::InvalidPacketId { packet_id: id });
781            return (self.take_events(), vec![]);
782        }
783        let mut reason = 0;
784        match self.reliability.received.get_mut(&id) {
785            Some((stage, channel))
786                if matches!(stage, ReceiveStage::PubRec | ReceiveStage::PubRel)
787                    && (*channel == stream || channel.is_none()) =>
788            {
789                *stage = ReceiveStage::PubRel;
790                *channel = stream;
791            }
792            None => {
793                reason = if self.mqtt_version() == 5 { 0x92 } else { 0 };
794                self.reliability
795                    .received
796                    .insert(id, (ReceiveStage::PubRel, stream));
797            }
798            _ => {
799                return (
800                    vec![MqttEvent::Error(MqttClientError::InvalidPacketId {
801                        packet_id: id,
802                    })],
803                    vec![],
804                )
805            }
806        }
807        let events = vec![MqttEvent::PubRelReceived {
808            packet_id: id,
809            stream,
810        }];
811        let responses = if self.options.auto_ack {
812            let packet = self
813                .manual_ack_packet(id, 7, reason, vec![], stream)
814                .expect("validated PUBREL state");
815            self.commit_receive_ack(id, 7, reason);
816            vec![packet]
817        } else {
818            vec![]
819        };
820        (events, responses)
821    }
822
823    pub(super) fn validate_ack(
824        &self,
825        packet: &MqttPacket,
826        stream: Option<u64>,
827    ) -> Result<(), MqttClientError> {
828        let (id, kind) = match packet {
829            MqttPacket::PubAck5(p) => (p.packet_id, 4),
830            MqttPacket::PubAck3(p) => (p.message_id, 4),
831            MqttPacket::PubRec5(p) => (p.packet_id, 5),
832            MqttPacket::PubRec3(p) => (p.message_id, 5),
833            MqttPacket::PubComp5(p) => (p.packet_id, 7),
834            MqttPacket::PubComp3(p) => (p.message_id, 7),
835            MqttPacket::SubAck5(p) => (p.packet_id, 9),
836            MqttPacket::SubAck3(p) => (p.message_id, 9),
837            MqttPacket::UnsubAck5(p) => (p.packet_id, 11),
838            MqttPacket::UnsubAck3(p) => (p.message_id, 11),
839            _ => return Ok(()),
840        };
841        let Some(entry) = self.inflight_queue.get(id) else {
842            return Err(MqttClientError::InvalidPacketId { packet_id: id });
843        };
844        let valid = match (&entry.packet, kind) {
845            (MqttPacket::Publish5(p), 4 | 5) => p.qos == if kind == 4 { 1 } else { 2 },
846            (MqttPacket::Publish3(p), 4 | 5) => p.qos == if kind == 4 { 1 } else { 2 },
847            (MqttPacket::PubRel5(_) | MqttPacket::PubRel3(_), 5 | 7) => true,
848            (MqttPacket::Subscribe5(_) | MqttPacket::Subscribe3(_), 9) => true,
849            (MqttPacket::Unsubscribe5(_) | MqttPacket::Unsubscribe3(_), 11) => true,
850            _ => false,
851        };
852        if !valid || entry.stream != stream {
853            return Err(MqttClientError::InvalidPacketId { packet_id: id });
854        }
855        let counts = match (&entry.packet, packet) {
856            (MqttPacket::Subscribe3(sent), MqttPacket::SubAck3(ack)) => {
857                Some((sent.subscriptions.len(), ack.return_codes.len()))
858            }
859            (MqttPacket::Subscribe5(sent), MqttPacket::SubAck5(ack)) => {
860                Some((sent.subscriptions.len(), ack.reason_codes.len()))
861            }
862            _ => None,
863        };
864        if let Some((expected, actual)) = counts {
865            if expected != actual {
866                return Err(MqttClientError::ProtocolViolation {
867                    message: format!(
868                        "SUBACK has {actual} results for {expected} subscription filters"
869                    ),
870                });
871            }
872        }
873        Ok(())
874    }
875
876    pub(super) fn fail_operation(&mut self, id: u16, packet: &MqttPacket, error: MqttClientError) {
877        let operation = match packet {
878            MqttPacket::Subscribe3(_) | MqttPacket::Subscribe5(_) => OperationKind::Subscribe,
879            MqttPacket::Unsubscribe3(_) | MqttPacket::Unsubscribe5(_) => OperationKind::Unsubscribe,
880            _ => OperationKind::Publish,
881        };
882        self.inflight_queue.acknowledge(id);
883        self.finish_operation(id);
884        self.events.push(MqttEvent::OperationFailed {
885            operation,
886            packet_id: Some(id),
887            error,
888        });
889    }
890
891    pub(super) fn process_replay(&mut self) {
892        if self.transport_manages_session {
893            return;
894        }
895        while let Some((id, packet)) = self.session_replay.front().cloned() {
896            let qos = match &packet {
897                MqttPacket::Publish5(p) => p.qos,
898                MqttPacket::Publish3(p) => p.qos,
899                _ => 2,
900            };
901            if matches!(packet, MqttPacket::Publish5(_) | MqttPacket::Publish3(_))
902                && !self.inflight_queue.can_push_publish()
903            {
904                break;
905            }
906            match self.enqueue_packet(packet.clone()) {
907                Ok(()) => {
908                    self.session_replay.pop_front();
909                    if let Err(error) = self.inflight_queue.push(id, packet.clone(), qos) {
910                        self.fail_operation(id, &packet, error);
911                    } else {
912                        self.start_deadline(OperationKind::Publish, Some(id));
913                    }
914                }
915                Err(MqttClientError::BufferFull { .. }) => break,
916                Err(error) => {
917                    self.session_replay.pop_front();
918                    self.fail_operation(id, &packet, error);
919                }
920            }
921        }
922    }
923}

925#[cfg(feature = "quic-proto")]
926impl QuicMqttEngine {
927    pub(super) fn check_stream_bytes(&self, additional: usize) -> Result<(), MqttClientError> {
928        if let Some(capacity) = self.mqtt_engine.options.max_outgoing_buffer_bytes {
929            let used = self.mqtt_engine.outgoing_bytes()
930                + self.control_outgoing.len()
931                + self
932                    .data_streams
933                    .values()
934                    .map(|s| s.outgoing.len())
935                    .sum::<usize>();
936            if used.saturating_add(additional) > capacity {
937                return Err(MqttClientError::BufferFull {
938                    buffer_type: "QUIC outgoing bytes".into(),
939                    capacity,
940                });
941            }
942        }
943        Ok(())
944    }
945
946    pub(super) fn preflight_publish(
947        &self,
948        command: &PublishCommand,
949    ) -> Result<(), MqttClientError> {
950        let mut command = command.clone();
951        self.mqtt_engine.validate_publish(&mut command)?;
952        if command.qos > 0 {
953            command.packet_id = Some(command.packet_id.unwrap_or(1));
954        }
955        let packet = if self.mqtt_engine.mqtt_version() == 5 {
956            MqttPacket::Publish5(command.to_mqtt_publish())
957        } else {
958            MqttPacket::Publish3(command.to_mqttv3_publish())
959        };
960        let bytes = packet.to_bytes().map_err(MqttClientError::from)?;
961        self.mqtt_engine.check_packet_size(&bytes)?;
962        self.check_stream_bytes(bytes.len())
963    }
964
965    pub(super) fn preflight_subscribe(
966        &self,
967        command: &SubscribeCommand,
968    ) -> Result<(), MqttClientError> {
969        self.mqtt_engine.validate_subscribe(command)?;
970        let id = command.packet_id.unwrap_or(1);
971        let packet = if self.mqtt_engine.mqtt_version() == 5 {
972            MqttPacket::Subscribe5(subscribev5::MqttSubscribe::new(
973                id,
974                command.subscriptions.clone(),
975                command.properties.clone(),
976            ))
977        } else {
978            MqttPacket::Subscribe3(subscribev3::MqttSubscribe::new(
979                id,
980                command
981                    .subscriptions
982                    .iter()
983                    .map(|s| subscribev3::SubscriptionTopic {
984                        topic_filter: s.topic_filter.clone(),
985                        qos: s.qos,
986                    })
987                    .collect(),
988            ))
989        };
990        let bytes = packet.to_bytes().map_err(MqttClientError::from)?;
991        self.mqtt_engine.check_packet_size(&bytes)?;
992        self.check_stream_bytes(bytes.len())
993    }
