8fn invalid(reason: &str) -> MqttClientError {
9    MqttClientError::InvalidConfiguration {
10        field: "session state".into(),
11        reason: reason.into(),
12    }
13}

15fn canonical(mut packet: MqttPacket) -> MqttPacket {
16    if let MqttPacket::Publish5(p) = &mut packet {
17        p.properties
18            .retain(|p| !matches!(p, Property::TopicAlias(_)));
19    }
20    packet
21}

23// Check these invariants even in builds without strict-protocol-compliance:
24// deserialized snapshots must not inject arbitrary packets or reuse live IDs.
25fn packet_id(
26    packet: &MqttPacket,
27    mqtt_version: u8,
28    queued: bool,
29) -> Result<Option<u16>, MqttClientError> {
30    let publish = match packet {
31        MqttPacket::Publish5(p) if mqtt_version == 5 => {
32            if p.properties
33                .iter()
34                .any(|p| matches!(p, Property::TopicAlias(_)))
35            {
36                return Err(invalid("Saved publications must not contain topic aliases"));
37            }
38            Some((p.qos, p.packet_id, &p.topic_name, p.dup))
39        }
40        MqttPacket::Publish3(p) if mqtt_version != 5 => {
41            Some((p.qos, p.message_id, &p.topic_name, p.dup))
42        }
43        _ => None,
44    };
45    if let Some((qos, id, topic, dup)) = publish {
46        if qos > 2
47            || topic.is_empty()
48            || topic.contains(['#', '+', '\0'])
49            || (qos == 0 && (id.is_some() || dup || !queued))
50            || (qos > 0 && !matches!(id, Some(1..=u16::MAX)))
51        {
52            return Err(invalid("Invalid saved publication"));
53        }
54        return Ok(id);
55    }
56    if queued {
57        return Err(invalid("Only publications may be queued"));
58    }
59    let id = match packet {
60        MqttPacket::PubRel5(p) if mqtt_version == 5 => p.packet_id,
61        MqttPacket::Subscribe5(p) if mqtt_version == 5 => p.packet_id,
62        MqttPacket::Unsubscribe5(p) if mqtt_version == 5 => p.packet_id,
63        MqttPacket::PubRel3(p) if mqtt_version != 5 => p.message_id,
64        MqttPacket::Subscribe3(p) if mqtt_version != 5 => p.message_id,
65        MqttPacket::Unsubscribe3(p) if mqtt_version != 5 => p.message_id,
66        _ => return Err(invalid("Invalid outstanding packet type or MQTT version")),
67    };
68    if id == 0 {
69        return Err(invalid("Saved packet identifiers must be nonzero"));
70    }
71    Ok(Some(id))
72}

74impl MqttEngine {
75    /// Capture an owned, serializable checkpoint without changing the engine.
76    ///
77    /// Requires a nonempty client ID and session tracking. Includes accepted
78    /// queued publications and outstanding QoS exchanges, including replay that
79    /// is waiting for quota or CONNACK. Connection-specific state is excluded.
80    ///
81    /// The application must write this through its `ClientSessionStore`. For a
82    /// planned restart, stop driving the engine while saving the final snapshot.
83    /// For crash recovery, save before exposing each transition's events/output.
84    /// `take_outgoing` also advances state: save after draining it and before
85    /// sending its bytes. On storage failure, keep output/events private and
86    /// retry or stop. Incoming application work must be durably accepted with
87    /// its checkpoint; application events themselves are not stored here.
88    pub fn snapshot_session(&self) -> Result<ClientSessionState, MqttClientError> {
89        if self.options.sessionless || self.options.client_id.is_empty() {
90            return Err(invalid(
91                "Checkpointing requires session tracking and a client ID",
92            ));
93        }
94        let mut outbound: Vec<_> = self
95            .previous_inflight
96            .iter()
97            .filter(|(id, _)| self.state.reserved.contains(id))
98            .cloned()
99            .collect();
100        let mut positions: HashMap<_, _> = outbound
101            .iter()
102            .enumerate()
103            .map(|(index, (id, _))| (*id, index))
104            .collect();
105        for (id, packet) in self
106            .inflight_queue
107            .snapshot_for_reconnect()
108            .into_iter()
109            .chain(self.session_replay.iter().cloned())
110        {
111            if let Some(&index) = positions.get(&id) {
112                outbound[index].1 = packet;
113            } else {
114                positions.insert(id, outbound.len());
115                outbound.push((id, packet));
116            }
117        }
118        let mut received: Vec<_> = self
119            .state
120            .received
121            .iter()
122            .filter_map(|(&(stream, packet_id), stage)| {
123                let stage = match stage {
124                    ReceiveStage::Publish(2) => ReceivedStage::Publish,
125                    ReceiveStage::PubRec => ReceivedStage::PubRec,
126                    ReceiveStage::PubRel => ReceivedStage::PubRel,
127                    _ => return None, // Incoming QoS 1 is connection-scoped.
128                };
129                Some(ReceivedExchange {
130                    stream,
131                    packet_id,
132                    stage,
133                })
134            })
135            .collect();
136        received.sort_by_key(|entry| (entry.stream, entry.packet_id));
137        Ok(ClientSessionState {
138            version: ClientSessionState::VERSION,
139            peer: self.options.peer.clone(),
140            client_id: self.options.client_id.clone(),
141            mqtt_version: self.options.mqtt_version,
142            has_session: self.state.has_session,
143            session_expiry_interval: self.state.session_expiry,
144            packet_id_counter: self
145                .session
146                .as_ref()
147                .map_or(0, ClientSession::packet_id_counter),
148            outbound: outbound
149                .into_iter()
150                .map(|(id, p)| (id, canonical(p)))
151                .collect(),
152            queued: self
153                .priority_queue
154                .iter()
155                .map(|(&priority, p)| (priority, canonical(p.clone())))
156                .collect(),
157            received,
158        })
159    }

161    /// Restore a checkpoint into a fresh engine, before CONNECT or commands.
162    ///
163    /// Requires the same peer, client ID and MQTT version, `clean_start(false)`
164    /// and session tracking. Supply credentials and other connection options
165    /// separately. A validation error leaves the engine unchanged. Reconnect
166    /// starts new timers, negotiates new limits and waits for CONNACK to decide
167    /// whether to replay or fail outstanding operations.
168    pub fn restore_session_state(
169        &mut self,
170        saved: ClientSessionState,
171    ) -> Result<(), MqttClientError> {
172        if self.session.is_some()
173            || self.is_connected
174            || self.state.connecting
175            || !self.state.reserved.is_empty()
176            || !self.priority_queue.is_empty()
177            || !self.outgoing_buffer.is_empty()
178        {
179            return Err(MqttClientError::InvalidState {
180                expected: "a fresh engine before CONNECT or commands".into(),
181                actual: "engine already in use".into(),
182            });
183        }
184        if saved.version != ClientSessionState::VERSION {
185            return Err(invalid("Unsupported session state version"));
186        }
187        if self.options.sessionless || self.options.clean_start {
188            return Err(invalid(
189                "Resumption requires session tracking and clean_start(false)",
190            ));
191        }
192        if saved.client_id.is_empty()
193            || saved.client_id != self.options.client_id
194            || saved.peer != self.options.peer
195            || saved.mqtt_version != self.options.mqtt_version
196            || !matches!(saved.mqtt_version, 3..=5)
197        {
198            return Err(invalid(
199                "Saved broker/client identity or MQTT version does not match",
200            ));
201        }
202        if saved.queued.len() > self.options.max_outgoing_packet_count {
203            return Err(invalid(
204                "Saved publications exceed the configured queue capacity",
205            ));
206        }
207        let mut reserved = HashSet::new();
208        let mut replay_reserve = 0;
209        for (id, packet) in &saved.outbound {
210            if packet_id(packet, saved.mqtt_version, false)? != Some(*id) || !reserved.insert(*id) {
211                return Err(invalid(
212                    "Duplicate or inconsistent outstanding packet identifier",
213                ));
214            }
215            replay_reserve = replay_reserve.max(packet.to_bytes()?.len());
216        }
217        let mut queue = PriorityQueue::new(self.options.max_outgoing_packet_count);
218        let mut queued_bytes = 0usize;
219        for (priority, packet) in saved.queued {
220            let id = packet_id(&packet, saved.mqtt_version, true)?;
221            if let Some(id) = id {
222                if !reserved.insert(id) {
223                    return Err(invalid("Duplicate queued packet identifier"));
224                }
225            }
226            let size = packet.to_bytes()?.len();
227            queued_bytes = queued_bytes.saturating_add(size);
228            if id.is_some() {
229                replay_reserve = replay_reserve.max(size);
230            }
231            queue.enqueue(priority, packet);
232        }
233        let control_reserve = self.connect_packet()?.to_bytes()?.len().max(6);
234        self.check_output_capacity(
235            queued_bytes.saturating_add(control_reserve.max(replay_reserve)),
236        )?;
237        let mut received = HashMap::new();
238        for entry in saved.received {
239            let stage = match entry.stage {
240                ReceivedStage::Publish => ReceiveStage::Publish(2),
241                ReceivedStage::PubRec => ReceiveStage::PubRec,
242                ReceivedStage::PubRel => ReceiveStage::PubRel,
243            };
244            if !saved.has_session
245                || entry.packet_id == 0
246                || received
247                    .insert((entry.stream, entry.packet_id), stage)
248                    .is_some()
249            {
250                return Err(invalid("Invalid or duplicate incoming QoS 2 exchange"));
251            }
252        }
253
254        // Commit only after the complete checkpoint has passed validation.
255        let mut session = ClientSession::new();
256        session.restore_packet_id_counter(saved.packet_id_counter);
257        self.session = Some(session);
258        self.previous_inflight = saved.outbound;
259        self.priority_queue = queue;
260        self.state.reserved = reserved;
261        self.state.queued_bytes = queued_bytes;
262        self.state.control_reserve = control_reserve;
263        self.state.replay_reserve = replay_reserve;
264        self.state.has_session = saved.has_session;
265        self.state.session_expiry = saved.session_expiry_interval;
266        self.state.received_unbound = received.keys().copied().collect();
267        self.state.received = received;
268        self.state.input_closed = true;
269        Ok(())
270    }
