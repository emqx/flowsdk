49            errors: AtomicU64::new(0),

68    pub fn record_error(&self, kind: ErrorKind) {
69        self.record_errors(kind, 1);
70    }
71
72    pub fn record_puback_no_match(&self, count: u64) {
73        self.puback_no_match.fetch_add(count, Ordering::Relaxed);
74    }
75
76    pub fn record_errors(&self, kind: ErrorKind, count: u64) {
77        if count == 0 {
78            return;
79        }
80        self.errors.fetch_add(count, Ordering::Relaxed);
81        match kind {
82            ErrorKind::Socket => {
83                self.socket_errors.fetch_add(count, Ordering::Relaxed);
84            }
85            ErrorKind::Connect => {
86                self.connect_errors.fetch_add(count, Ordering::Relaxed);
87            }
88            ErrorKind::Send => {
89                self.send_errors.fetch_add(count, Ordering::Relaxed);
90            }
91            ErrorKind::Receive => {
92                self.receive_errors.fetch_add(count, Ordering::Relaxed);
93            }
94            ErrorKind::MqttConnect => {
95                self.mqtt_errors.fetch_add(count, Ordering::Relaxed);
96                self.mqtt_connect_errors.fetch_add(count, Ordering::Relaxed);
97            }
98            ErrorKind::MqttSubscribe => {
99                self.mqtt_errors.fetch_add(count, Ordering::Relaxed);
100                self.mqtt_subscribe_errors
101                    .fetch_add(count, Ordering::Relaxed);
102            }
103            ErrorKind::MqttPublish => {
104                self.mqtt_errors.fetch_add(count, Ordering::Relaxed);
105                self.mqtt_publish_errors.fetch_add(count, Ordering::Relaxed);
106            }
107            ErrorKind::MqttClient => {
108                self.mqtt_errors.fetch_add(count, Ordering::Relaxed);
109                self.mqtt_client_errors.fetch_add(count, Ordering::Relaxed);
110            }
111            ErrorKind::MqttDisconnect => {
112                self.mqtt_errors.fetch_add(count, Ordering::Relaxed);
113                self.mqtt_disconnect_errors
114                    .fetch_add(count, Ordering::Relaxed);
115            }
116        };
117    }
118

122            received: self.messages_received.load(Ordering::Relaxed),

124            puback_no_match: self.puback_no_match.load(Ordering::Relaxed),
125            errors: self.errors.load(Ordering::Relaxed),
126            socket_errors: self.socket_errors.load(Ordering::Relaxed),
127            connect_errors: self.connect_errors.load(Ordering::Relaxed),
128            send_errors: self.send_errors.load(Ordering::Relaxed),
129            receive_errors: self.receive_errors.load(Ordering::Relaxed),
130            mqtt_errors: self.mqtt_errors.load(Ordering::Relaxed),
131            mqtt_connect_errors: self.mqtt_connect_errors.load(Ordering::Relaxed),
132            mqtt_subscribe_errors: self.mqtt_subscribe_errors.load(Ordering::Relaxed),
133            mqtt_publish_errors: self.mqtt_publish_errors.load(Ordering::Relaxed),
134            mqtt_client_errors: self.mqtt_client_errors.load(Ordering::Relaxed),
135            mqtt_disconnect_errors: self.mqtt_disconnect_errors.load(Ordering::Relaxed),
136            connected: self.clients_connected.load(Ordering::Relaxed),
137            subscribed: self.clients_subscribed.load(Ordering::Relaxed),
138            done: self.clients_done.load(Ordering::Relaxed),
139            elapsed: self.start_time.elapsed(),
140        }
141    }

166pub fn merge_latency_samples(worker_samples: Vec<Vec<Duration>>) -> Vec<Duration> {
167    let total: usize = worker_samples.iter().map(|v| v.len()).sum();
168    let mut merged = Vec::with_capacity(total);
169    for mut samples in worker_samples {
170        merged.append(&mut samples);
171    }
172    merged.sort_unstable();
173    merged
174}

176pub fn percentile(sorted: &[Duration], pct: f64) -> Duration {
177    if sorted.is_empty() {
178        return Duration::ZERO;
179    }
180    let idx = ((sorted.len() as f64 - 1.0) * pct / 100.0) as usize;
181    sorted[idx.min(sorted.len() - 1)]
182}

184pub fn print_final_summary(
185    config: &crate::config::BenchConfig,
186    snap: &StatsSnapshot,
187    latency_samples: &[Duration],
188) {
189    let duration_secs = snap.elapsed.as_secs_f64();
190    let messages = match config.action {
191        crate::config::BenchAction::Pub => snap.sent,
192        crate::config::BenchAction::Sub => snap.received,
193    };
194    let throughput = if duration_secs > 0.0 {
195        messages as f64 / duration_secs
196    } else {
197        0.0
198    };
199
200    let sep = "=".repeat(60);
201    let dash = "-".repeat(60);
202    println!("\n{sep}");
203    println!("{:^60}", "MQTT Bench Results");
204    println!("{sep}");
205    let transport = if config.quic { "QUIC" } else { "TCP" };
206    println!(
207        "  Broker:          {}:{} ({})",
208        config.host, config.port, transport
209    );
210    println!("  MQTT Version:    {}", config.mqtt_version);
211    println!("  Action:          {}", config.action.as_str());
212    println!("  Clients:         {}", config.clients);
213    println!("  Workers:         {}", config.workers);
214    println!("  QoS:             {}", config.qos);
215    match config.action {
216        crate::config::BenchAction::Pub => {
217            println!("  Payload Size:    {} bytes", config.payload_size);
218            println!("  Messages/Client: {}", config.messages);
219        }
220        crate::config::BenchAction::Sub => {
221            let target = if config.messages == 0 {
222                "unbounded".to_string()
223            } else {
224                config.messages.to_string()
225            };
226            println!("  Receive Target:  {} per client", target);
227            println!("  Subscribed:      {}", snap.subscribed);
228        }
229    }
230    println!("  Socket Buffers:  {} bytes", config.socket_buf);
231    println!("  Parser Buffer:   {} bytes", config.parser_buf);
232    println!("{dash}");
233    match config.action {
234        crate::config::BenchAction::Pub => {
235            println!("  Total Sent:      {}", snap.sent);
236            if config.qos > 0 {
237                println!("  Total Acked:     {}", snap.acked);
238                println!("  PubAckNoMatch:   {}", snap.puback_no_match);
239            }
240        }
241        crate::config::BenchAction::Sub => {
242            println!("  Total Received:  {}", snap.received);
243        }
244    }
245    println!("  Errors:          {}", snap.errors);
246    println!("    Socket:        {}", snap.socket_errors);
247    println!("    Connect:       {}", snap.connect_errors);
248    println!("    Send:          {}", snap.send_errors);
249    println!("    Receive:       {}", snap.receive_errors);
250    println!("    MQTT:          {}", snap.mqtt_errors);
251    println!("      CONNECT:     {}", snap.mqtt_connect_errors);
252    println!("      SUBSCRIBE:   {}", snap.mqtt_subscribe_errors);
253    println!("      PUBLISH:     {}", snap.mqtt_publish_errors);
254    println!("      Client:      {}", snap.mqtt_client_errors);
255    println!("      Disconnect:  {}", snap.mqtt_disconnect_errors);
256    println!("  Duration:        {:.2} s", duration_secs);
257    println!("  Throughput:      {:.2} msg/s", throughput);
258
259    if config.action == crate::config::BenchAction::Pub && !latency_samples.is_empty() {
260        println!("{dash}");
261        let label = if config.qos == 0 {
262            "Latency (enqueue time)"
263        } else {
264            "Latency (send -> ACK)"
265        };
266        println!("  {}:", label);
267        let avg: Duration = latency_samples.iter().sum::<Duration>() / latency_samples.len() as u32;
268        println!("    Min:           {:.2} ms", ms(latency_samples[0]));
269        println!(
270            "    Max:           {:.2} ms",
271            ms(*latency_samples.last().unwrap())
272        );
273        println!("    Avg:           {:.2} ms", ms(avg));
274        println!(
275            "    P50:           {:.2} ms",
276            ms(percentile(latency_samples, 50.0))
277        );
278        println!(
279            "    P95:           {:.2} ms",
280            ms(percentile(latency_samples, 95.0))
281        );
282        println!(
283            "    P99:           {:.2} ms",
284            ms(percentile(latency_samples, 99.0))
285        );
286        println!("    Samples:       {}", latency_samples.len());
287    }
288    println!("{sep}");
289}

295#[cfg(test)]
296mod tests {
297    use super::{BenchStats, ErrorKind};
298
299    #[test]
300    fn records_error_breakdown() {
301        let stats = BenchStats::new();
302        for kind in [
303            ErrorKind::Socket,
304            ErrorKind::Connect,
305            ErrorKind::Send,
306            ErrorKind::Receive,
307            ErrorKind::MqttConnect,
308            ErrorKind::MqttSubscribe,
309            ErrorKind::MqttPublish,
310            ErrorKind::MqttClient,
311            ErrorKind::MqttDisconnect,
312        ] {
313            stats.record_error(kind);
314        }
315        stats.record_puback_no_match(2);
316
317        let snapshot = stats.snapshot();
318        assert_eq!(snapshot.errors, 9);
319        assert_eq!(snapshot.puback_no_match, 2);
320        assert_eq!(snapshot.socket_errors, 1);
321        assert_eq!(snapshot.connect_errors, 1);
322        assert_eq!(snapshot.send_errors, 1);
323        assert_eq!(snapshot.receive_errors, 1);
324        assert_eq!(snapshot.mqtt_errors, 5);
325        assert_eq!(snapshot.mqtt_connect_errors, 1);
326        assert_eq!(snapshot.mqtt_subscribe_errors, 1);
327        assert_eq!(snapshot.mqtt_publish_errors, 1);
328        assert_eq!(snapshot.mqtt_client_errors, 1);
329        assert_eq!(snapshot.mqtt_disconnect_errors, 1);
330    }
